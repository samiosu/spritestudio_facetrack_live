cmake_minimum_required(VERSION 3.24)
project(SpriteFace VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
option(SPRITEFACE_BUILD_GUI "Build the native SpriteStudio preview application" ON)
option(SPRITEFACE_BUILD_TESTS "Build protocol and receiver tests" ON)
if(SPRITEFACE_BUILD_TESTS)
    # SDK regression tests are registered by the desktop subdirectory.
    enable_testing()
endif()
include(FetchContent)
find_package(Git REQUIRED)
find_package(Threads REQUIRED)
if(NOT WIN32)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
function(spriteface_verify_revision directory expected)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD
        RESULT_VARIABLE status OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(NOT status EQUAL 0 OR NOT revision STREQUAL expected)
        message(FATAL_ERROR "Dependency source differs from its audited commit: ${directory}. Expected ${expected}, got ${revision}.")
    endif()
endfunction()
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
spriteface_verify_revision("${json_SOURCE_DIR}" "9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03")
add_library(spriteface_core desktop/src/face.cpp desktop/src/receiver.cpp desktop/src/secure_packet.cpp)
target_include_directories(spriteface_core PUBLIC desktop/include)
target_link_libraries(spriteface_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
    target_link_libraries(spriteface_core PUBLIC ws2_32 bcrypt)
else()
    target_link_libraries(spriteface_core PUBLIC OpenSSL::Crypto)
endif()
if(MSVC)
    target_compile_options(spriteface_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(spriteface_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(spriteface-receiver desktop/src/receiver_main.cpp)
target_link_libraries(spriteface-receiver PRIVATE spriteface_core)
install(TARGETS spriteface-receiver RUNTIME DESTINATION bin)

    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)

    FetchContent_Declare(qrcodegen GIT_REPOSITORY https://github.com/nayuki/QR-Code-generator.git GIT_TAG v1.8.0 GIT_SHALLOW TRUE)
    FetchContent_Declare(spritestudio GIT_REPOSITORY https://github.com/SpriteStudio/SpriteStudio6-SDK.git GIT_TAG 2.3.1 GIT_SHALLOW TRUE)
    FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 2c980bb59875b0d32144a71867fbdebb2f77cd20)
    FetchContent_MakeAvailable(glfw imgui qrcodegen spritestudio stb)

    spriteface_verify_revision("${imgui_SOURCE_DIR}" "f5befd2d29e66809cd1110a152e375a7f1981f06")
    spriteface_verify_revision("${qrcodegen_SOURCE_DIR}" "720f62bddb7226106071d4728c292cb1df519ceb")
    spriteface_verify_revision("${spritestudio_SOURCE_DIR}" "8e5ea72d21eae5f143474691fbca05dacc45feb6")
    spriteface_verify_revision("${stb_SOURCE_DIR}" "2c980bb59875b0d32144a71867fbdebb2f77cd20")
